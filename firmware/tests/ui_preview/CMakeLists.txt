cmake_minimum_required(VERSION 3.20)
project(token_ui_preview C)
if(NOT DEFINED LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to LVGL 9.5.0")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "")
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(token_ui_preview preview.c ../../main/token_ui.c ../../main/token_model.c ../../main/postcard_codec.c ../../assets/fonts/token_font_14.c ../../assets/images/postcard_default_rgb565.c)
target_include_directories(token_ui_preview PRIVATE ../../main)
target_link_libraries(token_ui_preview PRIVATE lvgl)
